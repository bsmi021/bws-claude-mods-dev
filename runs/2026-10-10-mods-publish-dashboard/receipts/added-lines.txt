+Six [Claude Code mods](https://code.claude.com/docs/en/plugins/mods/overview) for game and app development on Windows. Each is a plugin of function hooks: JavaScript/TypeScript that runs inside Claude Code.
+| [`bws-cockpit`](bws-cockpit/README.md) | A dashboard pane beside the transcript with its own tabs: Assumptions (Claude records each assumption it makes through a tool and keeps working; mark each Right or Wrong with a correction and Send, and Claude fixes the work that relied on a wrong one), Shelf (an idea shelf for tangents: `/idea <text>` parks one, Claude parks its own through a tool, Resume / Done / Drop; one file shared by every session), Board (bws-board's live board) and Loop (harness loops: stall alerts, stop and resume, answering a paused loop's questions). New assumptions, an ask for this session and loop alerts bring their tab up; you can close it. A one-row action bar above the prompt: the newest run folder and its open items, live Unreal, Unity, Blender, adb and VRAM state, counts, and buttons for resume, the dashboard and its tabs, `/handoff`, shelving the draft and enhancing it. An Editor & Device Guard that blocks Unreal MCP, adb and Unity CLI calls that cannot work, and adds `-automated` and low-VRAM flags to editor launches. A Finish-Line Guard that keeps Claude working while the checklist of the run folder this session wrote to has open items. A Project Locator: a `find_project` tool and `/projects`, plus searches for `.uproject` and Unity project files answered from an index. `/handoff`, which writes a HANDOFF.md for a fresh agent and copies its start prompt (with a nudge when the context fills up), and a worker model pin for subagents. Since 0.15.0 it includes what were bws-assumptions and bws-shelf. |
+| [`bws-board`](bws-board/README.md) | A blackboard for several Claude Code sessions working in one repository, as peers. Each session has a card (role, task, busy or idle, claimed files). A file one session edits, or writes from Bash or PowerShell, is claimed, and another live session's write to it is refused before it runs. Sessions post asks to a role, a session or anyone (take, done, decline), send notes, and can wake an idle named peer for an urgent ask. `/board` prints the board; `/board pane` opens bws-cockpit's Board tab. One session at a time runs git that changes the tree, a commit that would sweep up a peer's claimed files is refused, and so are git commands that would overwrite a peer's changed files. Each session keeps a history of its board actions, and `/board digest` writes a self-contained timeline page of the team's work. |

+async function treeOf($: EngineInterface, ctx: Ctx, dir: string): Promise<Tree | null> {
+  const id = `${ctx.cwd}|${dir}`
+  const hit = trees.get(id)
+  if (hit !== undefined) return hit

+  const tree = top !== '' && normFile(common, '') === `${normFile(ctx.folder, '')}/.git` ? { top: toSlash(top), key: normFile(top, '') } : null
+  trees.set(id, tree)

+// The files a commit would take in that peers claim: the staged files, plus for -a the modified
+// tracked ones. An add or unstage earlier in the same command has not run yet, so its paths count:
+// an add -A/./-u counts every modified and untracked file, restore --staged or reset takes files out.
+// Runs git only when a peer claims a file in the tree.
+async function heldInCommit($: EngineInterface, o: Options, ctx: Ctx, b: Board, op: GitOp, tree: Tree, now: number) {
+  const peers = claimersIn(o, ctx, b, tree, now)
+  if (peers.length === 0) return []
+  const list = (args: string[]) => gitNames($, tree, args)
+  const st = op.staged
+  const added = (st?.paths ?? []).map(p => normFile(p, ctx.cwd))
+  const dropped = (st?.dropped ?? []).map(p => normFile(p, ctx.cwd))
+  // The index as it is now, less what the command unstages before its commit.
+  const index = st?.isReset === true ? new Set<string>() : await list(['diff', '--cached', '--name-only'])
+  const modified = op.isAll || st?.isAll === true ? await list(['diff', '--name-only']) : new Set<string>()
+  const untracked = st?.isAll === true ? await list(['ls-files', '--others', '--exclude-standard']) : new Set<string>()
+  const isIn = (f: string) => (index.has(f) && !under(f, dropped)) || modified.has(f) || untracked.has(f) || under(f, added)
+  return heldBy(o, ctx, b, peers, now, isIn)

+const under = (f: string, dirs: readonly string[]) => dirs.some(a => f === a || f.startsWith(`${a}/`))

+// Live peers with a claim in this tree: without one, the git checks run no git at all.

+// The peers' claimed files that `isIn` says a command reaches, with who holds each.
+function heldBy(o: Options, ctx: Ctx, b: Board, peers: readonly Card[], now: number, isIn: (file: string) => boolean) {
+  const held: { file: string; holder: Card }[] = []
+  for (const c of peers)
+    for (const cl of liveClaims(c.claims, now, o.lim.claimMs)) {
+      if (!isIn(cl.path)) continue
+      const holder = claimHolder(b, ctx.selfId, cl.path, now, o.lim)
+      if (holder !== undefined && !held.some(h => h.file === cl.path)) held.push({ file: cl.path, holder })
+    }
+  return held

+async function gitNamesOrFail($: EngineInterface, tree: Tree, args: string[]): Promise<Set<string> | undefined> {

+// The peers' claimed files a stash, hard reset, forced checkout, restore, clean or rm would overwrite
+// or delete. Only files git says would change count (see Clobber's scopes), so a claim on a file the
+// holder already committed never blocks. A git failure counts nothing: the command goes on.
+async function heldInClobber($: EngineInterface, o: Options, ctx: Ctx, b: Board, op: GitOp, c: Clobber, tree: Tree, now: number) {
+  const peers = claimersIn(o, ctx, b, tree, now)
+  if (peers.length === 0) return []
+  const named = c.paths.map(p => normFile(p, ctx.cwd))
+  const reaches = (f: string) => (c.isAll ? f.startsWith(`${tree.key}/`) : under(f, named))
+  // A delete or restore-from-a-commit of named files counts every claim on them; of a glob or a
+  // variable (the whole tree, as far as the board can tell), only the files git says changed.
+  if (c.scope === 'any' && !c.isAll) return heldBy(o, ctx, b, peers, now, reaches)
+  if (c.scope === 'clean') {
+    // git clean -n with the command's own arguments (less -f and -q) prints exactly what it would remove.
+    const args = (c.args ?? []).flatMap(a => (a === '--force' || a === '--quiet' ? [] : /^-[A-Za-z]+$/.test(a) ? [a.replace(/[fq]/g, '')].filter(x => x !== '-') : [a]))
+    const at = op.dir ? normFile(op.dir, ctx.cwd) : normFile(ctx.cwd, '')
+    const r = await settle($.process.run(['git', '-c', 'core.quotepath=off', 'clean', '-n', ...args], { cwd: op.dir || ctx.cwd, timeoutMs: 15_000 }))
+    if (r === undefined || r.exitCode !== 0) return []
+    const gone = r.stdout
+      .split(/\r?\n/)
+      .flatMap(l => /^Would remove (.+)$/.exec(l)?.[1] ?? [])
+      .map(n => normFile(n.replace(/\/$/, ''), at))
+    return heldBy(o, ctx, b, peers, now, f => under(f, gone))
+  }
+  const changed =
+    c.scope === 'index'
+      ? await gitNames($, tree, ['diff', '--name-only'])
+      : ((await gitNamesOrFail($, tree, ['diff', 'HEAD', '--name-only'])) ??
+        new Set([...(await gitNames($, tree, ['diff', '--cached', '--name-only'])), ...(await gitNames($, tree, ['diff', '--name-only']))]))
+  const untracked = c.untracked ? await gitNames($, tree, ['ls-files', '--others', '--exclude-standard']) : new Set<string>()
+  return heldBy(o, ctx, b, peers, now, f => reaches(f) && (changed.has(f) || untracked.has(f)))
+}

+// The git check for a shell command: refused while a peer holds the lease in the same tree, or when
+// a commit would take in a peer's claimed files; otherwise this session takes the lease.
+async function gitCheck($: EngineInterface, o: Options, ops: readonly GitOp[]): Promise<string | undefined> {
+  const ctx = await boardCtx($)
+  if (ctx === undefined) return undefined

+    if (holder !== undefined) {

+      return gitDenyText(holder, op.verb, now, o.lim.leaseMs)

+    // After a `cd` the parser could not follow, the command may run in another tree: not judged.

+      if (held.length > 0) {

+  if (!isFresh) void logEvent($, o, ctx, { kind: 'gitHold', at: now, verb: t.verb })
+  return undefined
+}

+  // The main loop's turn ended, so this session is idle (a subagent's turn is not) and its git lease
+  // is free. Awaited, so the card reads idle before the next prompt marks it busy.
+    if (!e.agentId) await settle(idle($, o))

+    if (ops.length > 0) {

+// ---------- git ----------
+
+// The git subcommands that change the index, the working tree's checkout or refs: one session at
+// a time runs these in a tree (the board's git lease). Read-only git (status, diff, log, show,
+// branch, fetch, tag, ...) is not here.
+export const GIT_LEASE_VERBS: ReadonlySet<string> = new Set([
+  'add', 'rm', 'mv', 'commit', 'push', 'pull', 'merge', 'rebase', 'reset', 'checkout', 'switch', 'restore', 'stash', 'cherry-pick', 'revert', 'am',
+])
+// git's global options that take the next word as their value.
+const GIT_VALUED = /^(?:-C|-c|--git-dir|--work-tree|--namespace|--config-env|--super-prefix)$/
+// commit's short options that take a value: in a cluster (-am), what follows one is that value.
+const COMMIT_VALUED = /[mFCctS]/
+
+// One git command that takes the lease. `dir`: where it runs, as written ('' = the session's own
+// folder; undefined = a `cd` the parser could not follow). `isAll`: commit -a, or add -A/-u/./a glob.
+// `paths`: add's plain paths, joined to `dir`. `unstage`: what restore --staged or a mixed reset takes
+// out of the index (`isAll`: all of it). A commit carries what the adds and unstages before it in the
+// same command leave staged (`staged`), since the check runs before any of the command does:
+// `paths` added, `dropped` taken out, `isAll` add -A and the like, `isReset` the index reset to HEAD.
+export type Staged = { isAll: boolean; paths: string[]; dropped: string[]; isReset: boolean }
+// What a git command overwrites or deletes in the working tree. `scope`: which of the files it reaches
+// it changes: 'head' those changed vs HEAD (staged or not), 'index' those changed vs the index, 'any'
+// every one (a delete, or a restore from another commit), 'clean' what `git clean -n` with the same
+// arguments lists. `isAll`: the whole tree; `paths`: the named ones, joined to `dir`; `untracked`:
+// stash -u/-a takes untracked files too; `args`: clean's own arguments, for its dry run.
+export type Clobber = { scope: 'head' | 'index' | 'any' | 'clean'; isAll: boolean; paths: string[]; untracked?: boolean; args?: string[] }
+// `isLease`: the command takes the git lease (git clean only clobbers).
+export type GitOp = {
+  verb: string
+  dir: string | undefined
+  isAll: boolean
+  paths: string[]
+  isLease: boolean
+  unstage?: { isAll: boolean; paths: string[] }
+  staged?: Staged
+  clobber?: Clobber
+}
+
+// The path arguments of a git subcommand: words that are not options (nor an option's value).
+const pathArgs = (args: string[], valued: RegExp) => {
+  const out: string[] = []
+  let isPaths = false
+  for (let i = 0; i < args.length; i++) {
+    const a = args[i] ?? ''
+    if (isPaths || !a.startsWith('-')) out.push(a)
+    else if (a === '--') isPaths = true
+    else if (valued.test(a)) i++
+  }
+  return out
+}
+
+const gitOp = (words: string[], base: string | undefined): GitOp | undefined => {
+  let dir = base
+  let i = 0
+  for (; i < words.length; i++) {
+    const w = words[i] ?? ''
+    if (!w.startsWith('-')) break
+    // Another git folder or work tree: not this board's to judge.
+    if (/^--(?:git-dir|work-tree)(?:=|$)/.test(w)) return undefined
+    if (GIT_VALUED.test(w)) {
+      const v = words[++i] ?? ''
+      if (w === '-C') dir = !isConcrete(v) ? undefined : dir === undefined ? (isAbsPath(v) ? v : undefined) : joinTo(dir, v)
+    }
+  }
+  const verb = words[i]
+  if (verb === undefined) return undefined
+  const args = words.slice(i + 1)
+  const clobber = clobberOf(verb, args, dir)
+  if (!GIT_LEASE_VERBS.has(verb)) return clobber === undefined ? undefined : { verb, dir, isAll: false, paths: [], isLease: false, clobber }
+  const op = leaseOp(verb, args, dir)
+  return op === undefined ? undefined : { ...op, isLease: true, ...(clobber ? { clobber } : {}) }
+}
+
+// A flag given long or inside a short cluster (-fd has f).
+const hasFlag = (args: readonly string[], long: RegExp, short: string) =>
+  args.some(a => long.test(a) || new RegExp(`^-[A-Za-z]*[${short}][A-Za-z]*$`).test(a))
+const isRefLike = (a: string) => /^(?:HEAD|@|FETCH_HEAD|ORIG_HEAD|MERGE_HEAD)$/.test(a) || /[~^]/.test(a) || /^[0-9a-f]{7,40}$/.test(a)
+// Named paths joined to `dir`; a pathspec magic, a glob or an unknown folder means the whole tree.
+const pathsIn = (named: readonly string[], dir: string | undefined) => {
+  const isAll = dir === undefined || named.some(p => p.startsWith(':') || !isConcrete(p))
+  return { isAll, paths: isAll || dir === undefined ? [] : named.map(p => joinTo(dir, p)) }
+}
+
+// What a git command overwrites or deletes in the working tree, if anything. Dry runs, stash
+// list/show/pop/apply/drop, rm --cached, a branch switch without -f and reset --merge/--keep are not
+// here: git itself refuses those when local changes would be lost.
+const clobberOf = (verb: string, args: string[], dir: string | undefined): Clobber | undefined => {
+  const whole: Clobber = { scope: 'head', isAll: true, paths: [] }
+  if (verb === 'reset') return args.includes('--hard') ? whole : undefined
+  if (verb === 'switch') return hasFlag(args, /^--(?:force|discard-changes)$/, 'f') ? whole : undefined
+  if (verb === 'checkout') {
+    const dashes = args.indexOf('--')
+    const flags = dashes < 0 ? args : args.slice(0, dashes)
+    const isForce = hasFlag(flags, /^--force$/, 'f')
+    const named: string[] = []
+    let isNewBranch = false
+    for (let k = 0; k < flags.length; k++) {
+      const a = flags[k] ?? ''
+      if (/^-(?:b|B)$|^--orphan$/.test(a)) {
+        isNewBranch = true
+        k++
+      } else if (/^--(?:conflict|pathspec-from-file)$/.test(a)) k++
+      else if (!a.startsWith('-')) named.push(a)
+    }
+    if (isForce) return whole
+    if (isNewBranch) return undefined
+    if (dashes >= 0) {
+      const after = args.slice(dashes + 1)
+      return after.length === 0 ? undefined : { scope: named.length > 0 ? 'any' : 'index', ...pathsIn(after, dir) }
+    }
+    if (named.length === 0) return undefined
+    // `checkout <ref> <paths>` restores from that commit; otherwise every name may be a path (a
+    // branch name matches no claimed file, so reading it as a path costs nothing).
+    const first = named[0] ?? ''
+    if (named.length > 1 && isRefLike(first)) return { scope: 'any', ...pathsIn(named.slice(1), dir) }
+    return { scope: 'index', ...pathsIn(named, dir) }
+  }
+  if (verb === 'restore') {
+    const isStaged = hasFlag(args, /^--staged$/, 'S')
+    const isWorktree = hasFlag(args, /^--worktree$/, 'W')
+    if (isStaged && !isWorktree) return undefined
+    const named = pathArgs(args, /^(?:-s|--source)$/)
+    if (named.length === 0) return undefined
+    const hasSource = args.some(a => a === '-s' || /^--source(?:=|$)/.test(a) || /^-s./.test(a))
+    return { scope: hasSource ? 'any' : isStaged ? 'head' : 'index', ...pathsIn(named, dir) }
+  }
+  if (verb === 'stash') {
+    const sub = args[0]
+    if (sub !== undefined && !sub.startsWith('-') && sub !== 'push' && sub !== 'save') return undefined
+    const rest = sub === 'push' || sub === 'save' ? args.slice(1) : args
+    const untracked = hasFlag(rest, /^--(?:include-untracked|all)$/, 'ua')
+    const named = sub === 'save' ? [] : pathArgs(rest, /^(?:-m|--message|--pathspec-from-file)$/)
+    return { ...(named.length === 0 ? whole : { scope: 'head' as const, ...pathsIn(named, dir) }), ...(untracked ? { untracked: true } : {}) }
+  }
+  if (verb === 'clean') {
+    if (!hasFlag(args, /^--force$/, 'f') || hasFlag(args, /^--(?:dry-run|interactive)$/, 'ni')) return undefined
+    return { scope: 'clean', isAll: true, paths: [], args: [...args] }
+  }
+  if (verb === 'rm') {
+    if (args.includes('--cached') || hasFlag(args, /^--dry-run$/, 'n')) return undefined
+    const named = pathArgs(args, /^--pathspec-from-file$/)
+    return named.length === 0 ? undefined : { scope: 'any', ...pathsIn(named, dir) }
+  }
+  return undefined
+}
+
+// The lease part of a git command (no `isLease` yet).
+const leaseOp = (verb: string, args: string[], dir: string | undefined): Omit<GitOp, 'isLease'> | undefined => {
+  if (verb === 'stash' && /^(?:list|show)$/.test(args[0] ?? '')) return undefined
+  if (verb === 'commit') {
+    const isAll = args.some(a => a === '--all' || (/^-[A-Za-z]+$/.test(a) && (a.slice(1).split(COMMIT_VALUED)[0] ?? '').includes('a')))
+    return { verb, dir, isAll, paths: [] }
+  }
+  if (verb === 'add') {
+    const dashes = args.indexOf('--')
+    const flags = dashes < 0 ? args : args.slice(0, dashes)
+    const named = [...flags.filter(a => !a.startsWith('-')), ...(dashes < 0 ? [] : args.slice(dashes + 1))]
+    const isAll =
+      dir === undefined ||
+      flags.some(a => /^(?:-A|-u|--all|--update|--no-ignore-removal|--pathspec-from-file(?:=.*)?)$/.test(a) || /^-[A-Za-z]*[Au][A-Za-z]*$/.test(a)) ||
+      named.some(p => p === '.' || p.startsWith(':') || !isConcrete(p))
+    return { verb, dir, isAll, paths: isAll || dir === undefined ? [] : named.map(p => joinTo(dir, p)) }
+  }
+  if (verb === 'restore' && args.some(a => a === '--staged' || /^-[A-Za-z]*S[A-Za-z]*$/.test(a))) {
+    const named = pathArgs(args, /^(?:-s|--source)$/)
+    const isAll = dir === undefined || named.length === 0 || named.some(p => p === '.' || p.startsWith(':') || !isConcrete(p))
+    return { verb, dir, isAll: false, paths: [], unstage: { isAll, paths: isAll || dir === undefined ? [] : named.map(p => joinTo(dir, p)) } }
+  }
+  // A mixed reset (the default) unstages: `reset`, `reset HEAD`, `reset [HEAD] [--] <paths>`.
+  if (verb === 'reset' && !args.some(a => /^--(?:hard|soft|merge|keep)$/.test(a))) {
+    const dashes = args.indexOf('--')
+    const before = (dashes < 0 ? args : args.slice(0, dashes)).filter(a => !a.startsWith('-'))
+    const after = dashes < 0 ? [] : args.slice(dashes + 1)
+    const isRef = (a: string | undefined) => a !== undefined && /^(?:HEAD|@)(?:[~^]\d*)*$/.test(a)
+    const named = [...(isRef(before[0]) ? before.slice(1) : before), ...after]
+    // `reset <commit>` moves the branch: not an unstage this check can follow.
+    if (dashes < 0 && before.length === 1 && !isRef(before[0]) && !/[./]/.test(before[0] ?? '')) return { verb, dir, isAll: false, paths: [] }
+    const isAll = dir === undefined || named.length === 0 || named.some(p => p === '.' || p.startsWith(':') || !isConcrete(p))
+    return { verb, dir, isAll: false, paths: [], unstage: { isAll, paths: isAll || dir === undefined ? [] : named.map(p => joinTo(dir, p)) } }
+  }
+  return { verb, dir, isAll: false, paths: [] }
+}
+
+// What the adds and unstages, in order, leave in the index on top of what it held.
+const stagedBy = (ops: readonly GitOp[]): Staged => {
+  let st: Staged = { isAll: false, paths: [], dropped: [], isReset: false }
+  for (const o of ops) {
+    if (o.verb === 'add' && o.isAll) st = { ...st, isAll: true, dropped: [] }
+    else if (o.verb === 'add') st = { ...st, paths: [...st.paths, ...o.paths], dropped: st.dropped.filter(d => !o.paths.includes(d)) }
+    else if (o.unstage?.isAll) st = { isAll: false, paths: [], dropped: [], isReset: true }
+    else if (o.unstage) st = { ...st, paths: st.paths.filter(p => !o.unstage?.paths.includes(p)), dropped: [...st.dropped, ...o.unstage.paths] }
+  }
+  return st
+}
+
+// The lease-taking git commands in a shell command, command position only, in order. `cd` moves
+// where later ones run, as for writeTargets. A `git -C <dir>` names its own folder.
+export const gitOps = (cmd: string, dialect: Dialect): GitOp[] => {
+  const out: GitOp[] = []
+  let base: string | undefined = ''
+  for (const seg of splitSegments(cmd, dialect)) {
+    const inv = invocationOf(cmd, seg, dialect)
+    if (inv === undefined || inv.isStartProcess) continue
+    const args = lexSegment(inv.args, dialect).words.map(w => w.text)
+    if (CD.test(inv.base)) {
+      const dir = args.find(x => !x.startsWith('-'))
+      if (dir === undefined || !isConcrete(dir)) base = undefined
+      else if (base !== undefined) base = joinTo(base, dir)
+      else if (isAbsPath(dir)) base = dir
+      continue
+    }
+    if (!/^git(?:\.exe)?$/i.test(inv.base)) continue
+    const op = gitOp(args, base)
+    if (op === undefined) continue
+    if (op.verb === 'commit') {
+      const before = out.filter(o => o.dir === op.dir && (o.verb === 'add' || o.unstage !== undefined))
+      if (before.length > 0) op.staged = stagedBy(before)
+    }
+    out.push(op)
+  }
+  return out
+}

+// Writes HANDOFF.md into this session's run folder (as the Finish-Line Guard sees it), never one a worker freshened.
+async function writeHandoff($: EngineInterface, focus: string): Promise<string> {
+  const cwd = await $.session.cwd()
+  const now = await $.clock.now()
+  const run = await sessionRunFolder($)
+  const folder = run !== null ? run.folder : handoffFolder(now)
+  $.ui.toast(`handoff: writing ${folder}/HANDOFF.md`, { timeoutMs: 4000 })
+  let r = await $.model.fork({ prompt: handoffPrompt(folder, focus) })
+  // A resumed session has a transcript but nothing to fork until its first request: build from the rows instead.
+  if (!r.isAnswered && r.reason === 'nothing-to-fork') {
+    const rows = (await settle($.session.messages())) ?? []
+    const transcript = transcriptText(rows)
+    if (transcript.length === 0) return 'No handoff written: there is nothing to hand off yet: this conversation has no messages.'
+    r = await $.model.complete({ model: await $.session.model(), prompt: transcriptPrompt(transcript, folder, focus), maxTokens: 3000 })
+  }
+  if (!r.isAnswered) return `No handoff written: the model call failed (${r.reason}).`
+  const body = r.text.trim()
+  if (body.length === 0) return 'No handoff written: the model returned an empty reply.'
+  const path = joinPath(cwd, `${folder}/HANDOFF.md`)
+  const previous = await settle($.fs.read(path))
+  const kept = previous !== undefined ? keptName(now) : null
+  if (kept !== null) await $.fs.write(joinPath(cwd, `${folder}/${kept}`), previous ?? '')
+  await $.fs.write(path, `${body}\n`)
+  await update($, handoffNudge, h => (h === null ? h : { ...h, isDone: true }))
+  const start = startPrompt(folder)
+  const surfaces = await $.session.surfaces()
+  const copied = surfaces.length > 0 && (await settle($.ui.copy({ text: start })))?.isCopied === true
+  return (
+    `Handoff written: ${folder}/HANDOFF.md (${words(body)} words)` +
+    (kept !== null ? `; the previous one is kept as ${folder}/${kept}` : '') +
+    `.\nStart the next session with this prompt${copied ? ' (copied to the clipboard)' : ''}:\n${start}`

+export type Verdict = 'open' | 'right' | 'wrong'
+
+// One assumption Claude recorded, numbered A1, A2… in the order it arrived.
+export type Assumption = {
+  n: number
+  text: string
+  why: string
+  verdict: Verdict
+  // What it should be instead; only read when verdict is 'wrong'. Empty: Claude is told to ask.
+  correction: string
+  // True once its verdict went to Claude.
+  isSent: boolean
+}
+
+// An idea on the shelf, as hooks/shelf.ts writes it to the shelf file.
+export type ShelfIdea = {
+  id: number
+  text: string
+  detail?: string
+  by: 'user' | 'claude'
+  project: string
+  createdAt: number
+  status: 'open' | 'done' | 'dropped'
+  closedAt?: number
+}
+
+// What the Shelf tab draws: the open ideas as last loaded from the shelf file, and this project's open count.
+export type ShelfView = {
+  isAll: boolean
+  root: string
+  ideas: ShelfIdea[]
+  now: number
+  note: string
+  count: number
+}
+
+export type DashTab = 'assumptions' | 'shelf' | 'board' | 'loop'
+
+export type DashView = {
+  // The tab shown; null until something picks one.
+  tab: DashTab | null
+  // The person closed the dashboard: news lights the bar instead of reopening it, until they open it again.
+  isClosedByPerson: boolean
+}
+
+// The handoff nudge, once per session (keyed by its start): Handoff stays highlighted until it runs.
+export type HandoffNudge = { startedAt: number; isDone: boolean }
+
+      // the board as last read, drawn by the dashboard's Board tab
+      lines: string[]
+      assumptions: Assumption[]
