+## Hotkey
+The terminal footer carries a small `◨ side` button. Press it, or a chord bound to it, to open the side chat (the one that is there, or an empty one) and take the keyboard, with no command typed. A mod cannot bind a chord itself; the button borrows one of Claude Code's keybinding actions (`chordAction`, default `app:cycleDiffBase`, which only the diff pane handles), and you bind a chord to that action. Add this to `~/.claude/keybindings.json` on each computer:
+
+```json
+{
+  "bindings": [
+    { "context": "Global", "bindings": { "ctrl+x ctrl+o": "app:cycleDiffBase" } }
+  ]
+}
+```
+
+(merge it into the file if you already have bindings). `ctrl+x ctrl+o` is free in Claude Code 2.1.296; any chord works. While the `/diff` pane is open the chord belongs to it. With text in the prompt box the pane opens but the keyboard stays with you (`ctrl+x tab` or a click moves it). `chordAction` empty hides the button.
+
+## Install on another computer
+The repository is private, so the computer needs GitHub access first (`gh auth login`, or an SSH key). Then:
+
+```bash
+claude plugin marketplace add bsmi021/bws-claude-mods
+claude plugin install bws-sidechat@bws-mods
+```

+// The hotkey and the footer button: open the side chat (continuing the one there is), or take the keys back to it.
+async function summon($: EngineInterface) {
+  if (!(await isDrawn($))) return
+  await openPane($)
+}
+
+  const action = String(options.chordAction ?? '').trim()
+  // The hotkey. A mod cannot bind a chord itself, but a Button can borrow one of the engine's keybinding actions: the
+  // footer carries a small "side" button, and the chord the person binds to `chordAction` in ~/.claude/keybindings.json
+  // (Global) presses it from the prompt. It opens the pane, or takes the keys back to it, with no command typed.
+  on('ui.render', { component: 'SessionMode' }, async ($, e, next) => {
+    if (action === '' || e.surface !== 'terminal') return next(e)
+    // The footer holds one tree: share it. Another mod's footer (whatever next(e) draws) stays beside ours; with none,
+    // the engine's own mode labels are drawn here, as the engine would.
+    const drawn = await settle(next(e))
+    const { Box, Button, Text } = $.ui.resolve(e)
+    const other = drawn !== undefined && drawn.type !== 'engine' ? drawn : null
+    return (
+      <Box flexDirection="row" gap={1}>
+        {other ?? (e.props.modes.length > 0 ? <Text dimColor>{e.props.modes.join(' & ')}</Text> : null)}
+        <Button key="side" plain dimColor label="◨ side" action={action} onPress={() => void summon($)} />
+      </Box>
+    )
+  })
+
+const footer = (surface: 'terminal' | 'desktop' = 'terminal', modes: string[] = ['plan']) =>
+  ({ plugin: 'bws-sidechat', component: 'SessionMode', surface, props: { modes } }) as const
+
+describe('hotkey', () => {
+  test('the footer carries a side button on the chord action, beside the engine mode labels', async ($, on) => {
+    world(on)
+    const ui = await $.ui.mount(footer())
+    const button = await ui.find({ key: 'side' })
+    expect(button).toBeDefined()
+    expect(button?.props?.action).toBe('app:cycleDiffBase')
+    expect(String(button?.props?.label)).toContain('side')
+    expect(await ui.find({ type: 'Text', text: 'plan' })).toBeDefined()
+    await ui.unmount()
+  })
+
+  test('pressing it opens the pane with no command typed, and takes the keys', async ($, on) => {
+    const { seen, clock } = world(on)
+    const ui = await $.ui.mount(footer())
+    expect(seen.opened).toEqual([])
+    await ui.press({ key: 'side' })
+    expect(seen.opened[0]).toEqual({ id: 'sidechat', focus: undefined })
+    await clock.advance(100)
+    expect(seen.opened.filter(o => o.focus === true)).toHaveLength(1)
+    expect(seen.spawns).toEqual([]) // nothing is asked of the model by opening it
+    await ui.unmount()
+  })
+
+  test('it brings back the side chat that is there, not a new one', async ($, on) => {
+    const { seen, clock } = world(on)
+    await run($, 'btw', 'first question')
+    await clock.advance(200)
+    await run($, 'side', 'close')
+    const ui = await $.ui.mount(footer())
+    await ui.press({ key: 'side' })
+    await clock.advance(100)
+    await ui.unmount()
+    const pane1 = await $.ui.mount(PANE)
+    expect(await pane1.find({ type: 'Text', text: 'first question' })).toBeDefined()
+    await pane1.unmount()
+    expect(seen.spawns).toHaveLength(1)
+  })
+
+  test('with text in the prompt the pane opens but the keys stay with the person', async ($, on) => {
+    const { seen, clock } = world(on, { box: 'half a thought' })
+    const ui = await $.ui.mount(footer())
+    await ui.press({ key: 'side' })
+    await clock.advance(1000)
+    expect(seen.opened).toEqual([{ id: 'sidechat', focus: undefined }])
+    await ui.unmount()
+  })
+
+  test('chordAction picks the engine action', { options: { chordAction: 'app:toggleDiffNoiseFilter' } }, async ($, on) => {
+    world(on)
+    const ui = await $.ui.mount(footer())
