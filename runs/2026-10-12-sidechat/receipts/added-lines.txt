+# bws-sidechat
+
+A side chat in the right-hand pane. `/fork`, `/branch` and `/btw` open a copy of the current conversation beside it; you ask in the pane, the main conversation is never touched.
+
+## How it works
+Each side message runs a headless `claude -p`. The first one forks the main session (`--resume <main> --fork-session --session-id <new>`); later ones resume the fork. So the side chat has the full context, history included, and its own tools. Streamed text shows live in the pane.
+
+The child runs with `CLAUDE_CODE_PLUGIN_DIRS` emptied (faster start, no side-of-side) and, by default, no MCP servers (`--strict-mcp-config`). Tools it may use without asking: `Read,Glob,Grep,WebFetch,WebSearch`. It cannot ask permission, so anything else is denied.
+
+## Commands
+| Command | Does |
+|---|---|
+| `/fork`, `/branch` | (replaces the built-in background-agent fork and branch) New side chat forked from the conversation as it stands. Text after it is sent as the first message. `/fork native` runs the built-in fork. |
+| `/btw <question>` | Answer in the side chat. Continues an open side chat; forks if none. |
+| `/side [new [msg] \| close \| msg]` | The plugin's own entry point. |
+
+The pane has New fork, Close, Stop, an Ask box, and ↑/↓ buttons to scroll long chats. In headless runs the built-in commands are left alone.
+
+## Options (`/plugin` config)
+`takeOverFork`, `takeOverBtw` (off = built-in command), `sideTools`, `sideModel`, `loadMcpServers`, `claudePath`.
+
+## Notes
+- The fork sees the main session as last written to disk, so a turn still in flight is seen up to its last saved message.
+- Each fork is a real session file (`claude --resume <id>` opens it). It may also appear on the session board.
+- Tested on Claude Code 2.1.296, Windows 11, Windows Terminal at 120 columns: validation, 25 automated tests, and by hand `/btw`, `/fork` and `/side` opening the pane beside another pane and answering from the fork. Not checked by hand: typing in the Ask box, Stop, the scroll buttons, narrow terminals, the desktop app.

+// One side turn: the message goes to a headless `claude -p` that forks (first turn) or resumes (later ones) the side
+// session. The main conversation is never written to.
+async function send($: EngineInterface, text: string, cfg: SideConfig & { exe: string }) {
+  const message = text.trim()
+  if (message === '') return
+  let c = await read($, chat)
+  if (c.isBusy) return void $.ui.toast('Side chat: still answering. Stop it first, or wait.')
+  if (c.sideId === null || c.mainId === null) {
+    await newFork($)
+    c = await read($, chat)
+  }
+  const mainId = c.mainId as string
+  const sideId = c.sideId as string
+  const mine = ++runNo
+  isStopped = false
+  isInitSeen = false
+  const isFirst = !c.isForked
+  let draft: Draft = { msgs: [...c.msgs, { role: 'user', text: message }], live: '' }
+  await update($, chat, old => ({ ...old, ...draft, isBusy: true, back: 0 }))
+
+  let pending = ''
+  let stderr = ''
+  let code: number | null = 0
+  const stream = $.process.spawn({
+    argv: [cfg.exe, ...sideArgs({ message, mainId, sideId, isFirst }, cfg)],
+    // No mods in the child: it starts faster, and cannot open a side chat of its own.
+    env: { CLAUDE_CODE_PLUGIN_DIRS: '' },
+  })
+  try {
+    for (;;) {
+      const step = await stream.next()
+      if (mine !== runNo || isStopped) break
+      if (step.done) {
+        code = step.value.code
+        break
+      }
+      if (step.value.stream === 'stderr') {
+        stderr += step.value.text
+        continue
+      }
+      const split = splitLines(pending, step.value.text)
+      pending = split.rest
+      if (split.lines.length === 0) continue
+      for (const line of split.lines) {
+        if (line.includes('"subtype":"init"')) isInitSeen = true
+        draft = reduceLine(draft, line)
+      }
+      const snap = draft
+      await update($, chat, old => (mine === runNo ? { ...old, msgs: snap.msgs, live: snap.live } : old))
+    }
+  } catch (err) {
+    code = -1
+    stderr += String(err)
+  } finally {
+    await settle(stream.return(undefined as never))
+  }
+  if (mine !== runNo) return // a new fork or a close took over
+  const note = isStopped ? 'Stopped.' : exitNote(code, stderr)
+  const extra: SideMsg[] = note === null ? [] : [{ role: 'note', text: note }]
+  const shown = draft.live.trim() === '' || isStopped ? [] : [{ role: 'assistant', text: draft.live.trim() } as SideMsg]
+  await update($, chat, old => ({
+    ...old,
+    msgs: [...draft.msgs, ...shown, ...extra],
+    live: '',
+    isBusy: false,
+    isForked: old.isForked || isInitSeen || code === 0,
+  }))
+}
+
+async function stop($: EngineInterface) {
+  const c = await read($, chat)
+  if (!c.isBusy) return
+  isStopped = true
+  runNo++
+  await update($, chat, (old): SideChat => ({
+    ...old,
+    msgs: [...old.msgs, ...(old.live.trim() === '' ? [] : [{ role: 'assistant', text: old.live.trim() } as SideMsg]), { role: 'note', text: 'Stopped.' } as SideMsg],
+    live: '',
+    isBusy: false,
+    isForked: old.isForked || isInitSeen,
+  }))

+    const newer = c.msgs.length - win.to

+export type SideRun = {
+  message: string
+  mainId: string // the conversation forked from
+  sideId: string // the side session's own id, minted here so no init line has to be parsed
+  isFirst: boolean // first turn forks mainId; later turns resume sideId

+// `claude -p` for one side turn. The first turn forks the main session into sideId; later ones continue it.
+export function sideArgs(run: SideRun, cfg: SideConfig): string[] {
+  const args = ['-p', run.message]
+  if (run.isFirst) args.push('--resume', run.mainId, '--fork-session', '--session-id', run.sideId)
+  else args.push('--resume', run.sideId)
+  args.push('--output-format', 'stream-json', '--verbose', '--include-partial-messages')
+  if (cfg.isMcpOff) args.push('--strict-mcp-config')
+  const tools = cfg.tools.split(/[\s,]+/).filter(t => t !== '')
+  if (tools.length > 0) args.push('--allowedTools', ...tools)
+  if (cfg.model.trim() !== '') args.push('--model', cfg.model.trim())
+  return args
+}

+// Text arrives in pieces that end wherever a write did: whole lines out, the unfinished tail kept.
+export function splitLines(pending: string, chunk: string): { lines: string[]; rest: string } {
+  const parts = (pending + chunk).split('\n')
+  const rest = parts.pop() ?? ''
+  return { lines: parts.map(l => l.replace(/\r$/, '')).filter(l => l.trim() !== ''), rest }

+export type Draft = { msgs: SideMsg[]; live: string }

+const isObj = (v: unknown): v is Record<string, unknown> => typeof v === 'object' && v !== null
+
+// One short line for a tool the side chat ran: `Read C:/x/y.ts`.
+export function toolNote(name: string, input: unknown): string {
+  const o = isObj(input) ? input : {}
+  const arg = [o.file_path, o.path, o.pattern, o.command, o.url, o.query].find(v => typeof v === 'string')
+  const text = typeof arg === 'string' ? arg.replace(/\s+/g, ' ') : ''
+  return text === '' ? name : `${name} ${text.length > 70 ? text.slice(0, 69) + '…' : text}`
+}

+// Folds one stream-json line into the draft. Streamed text builds `live`; the finished assistant message replaces it.
+export function reduceLine(d: Draft, line: string): Draft {
+  let o: unknown
+  try {
+    o = JSON.parse(line)
+  } catch {
+    return d
+  }
+  if (!isObj(o)) return d
+  if (o.type === 'stream_event' && isObj(o.event)) {
+    const ev = o.event
+    const delta = isObj(ev.delta) ? ev.delta : null
+    if (ev.type === 'content_block_delta' && delta !== null && delta.type === 'text_delta' && typeof delta.text === 'string') {
+      return { msgs: d.msgs, live: d.live + delta.text }
+    }
+    return d
+  }
+  if (o.type === 'assistant' && isObj(o.message) && Array.isArray(o.message.content)) {
+    const added: SideMsg[] = []
+    const text = o.message.content
+      .filter((b): b is Record<string, unknown> => isObj(b) && b.type === 'text' && typeof b.text === 'string')
+      .map(b => String(b.text))
+      .join('')
+      .trim()
+    if (text !== '') added.push({ role: 'assistant', text })
+    for (const b of o.message.content) {
+      if (isObj(b) && b.type === 'tool_use' && typeof b.name === 'string') added.push({ role: 'tool', text: toolNote(b.name, b.input) })
+    }
+    return { msgs: [...d.msgs, ...added], live: '' }
+  }
+  if (o.type === 'result') {
+    const out: SideMsg[] = []
+    if (o.is_error === true) out.push({ role: 'note', text: `Error: ${typeof o.result === 'string' && o.result !== '' ? o.result : String(o.subtype ?? 'failed')}` })
+    else if (typeof o.result === 'string' && o.result.trim() !== '' && !d.msgs.some(m => m.role === 'assistant' && m.text === o.result)) {
+      // no assistant line carried it (a slash command's output, say)
+      if (d.live.trim() === '') out.push({ role: 'assistant', text: o.result.trim() })
+    }
+    return { msgs: [...d.msgs, ...out], live: '' }
+  }
+  return d
+}

+// What shows when the child exits without having printed a result (a missing exe, a bad flag).
+export function exitNote(code: number | null, stderr: string): string | null {
+  if (code === 0) return null
+  const tail = stderr.trim().split('\n').slice(-3).join(' ').slice(0, 300)
+  return `The side chat process ended (${code === null ? 'killed' : `exit ${code}`})${tail === '' ? '' : `: ${tail}`}`
+}

+export const ROLE_PREFIX: Record<SideMsg['role'], string> = { user: 'You', assistant: 'Claude', tool: '·', note: '!' }

+// Rows a message takes at this width (one header row, then its wrapped lines).
+export function rowsOf(m: SideMsg, width: number): number {
+  const w = Math.max(10, width)
+  if (m.role === 'tool' || m.role === 'note') return Math.max(1, Math.ceil((m.text.length + 2) / w))
+  return 1 + m.text.split('\n').reduce((n, l) => n + Math.max(1, Math.ceil(l.length / w)), 0)
+}
+
+// The newest messages that fit `rows`, skipping `back` messages from the end (the person scrolled up).
+export function windowOf(msgs: readonly SideMsg[], rows: number, width: number, back: number): { from: number; to: number } {
+  const to = Math.max(0, msgs.length - Math.max(0, back))
+  let used = 0
+  let from = to
+  while (from > 0) {
+    const need = rowsOf(msgs[from - 1] as SideMsg, width)
+    if (used + need > rows && from < to) break
+    used += need
+    from--
+  }
+  return { from, to }
+}

+export function nativeArgs(args: string): string | null {
+  const m = /^\s*native(?:\s+([\s\S]*))?$/i.exec(args)
+  return m === null ? null : (m[1] ?? '').trim()

+export type SideCommand = { kind: 'open' } | { kind: 'new'; message: string } | { kind: 'close' } | { kind: 'send'; message: string }

+// `/side [new [msg] | close | msg]`
+export function parseSide(args: string): SideCommand {
